**`client/src/lib/markdown-section-preview-cache.ts`**

```typescript
/**
 * Section HTML 캐시 자료구조.
 * section ID → sanitized HTML 맵과 현재 section 순서를 유지한다.
 */
export class SectionPreviewCache {
  /** section ID → sanitized HTML */
  private readonly htmlCache = new Map<string, string>();
  /** 현재 section 순서 */
  private sectionOrder: string[] = [];

  /**
   * 현재 section 순서를 반환한다.
   *
   * @returns section ID 배열 (순서 보장)
   */
  getSectionOrder(): string[] {
    return [...this.sectionOrder];
  }

  /**
   * section 순서를 갱신한다.
   * 순서가 변경되면 true를 반환한다.
   *
   * @param newOrder 새 section ID 배열
   * @returns 순서 변경 여부
   */
  updateSectionOrder(newOrder: string[]): boolean {
    const changed =
      newOrder.length !== this.sectionOrder.length ||
      newOrder.some((id, i) => id !== this.sectionOrder[i]);
    if (changed) {
      this.sectionOrder = [...newOrder];
    }
    return changed;
  }

  /**
   * section HTML을 캐시에 저장한다.
   *
   * @param sectionId section ID
   * @param html sanitized HTML
   */
  setHtml(sectionId: string, html: string): void {
    this.htmlCache.set(sectionId, html);
  }

  /**
   * section HTML을 캐시에서 조회한다.
   *
   * @param sectionId section ID
   * @returns 캐시된 HTML (없으면 undefined)
   */
  getHtml(sectionId: string): string | undefined {
    return this.htmlCache.get(sectionId);
  }

  /**
   * 전체 캐시를 무효화한다.
   */
  invalidateAll(): void {
    this.htmlCache.clear();
  }

  /**
   * 현재 section 순서대로 HTML을 조합하여 반환한다.
   * 캐시가 비어 있는 section은 빈 문자열로 처리한다.
   *
   * @returns 전체 프리뷰 HTML
   */
  buildFullHtml(): string {
    return this.sectionOrder.map((id) => this.htmlCache.get(id) ?? '').join('');
  }
}
```

// 기존 메시지 타입 (하위 호환)
interface FullPreviewRequest { id: number; body: string; }
// 신규 메시지 타입 (section 단위)
interface SectionPreviewRequest { id: number; sectionId: string; sectionText: string; }

// onmessage에서 분기:
self.addEventListener('message', (event) => {
  const data = event.data as FullPreviewRequest | SectionPreviewRequest;
  if ('sectionId' in data) {
    // section 단위 렌더링
    const html = renderMarkdownPreview(data.sectionText);
    self.postMessage({ id: data.id, sectionId: data.sectionId, html });
  } else {
    // 전체 body 렌더링 (기존 호환)
    const html = renderMarkdownPreview(data.body);
    self.postMessage({ id: data.id, html });
  }
});

핵심 구현 사항:
1. `SectionPreviewCache` 인스턴스를 `useRef`로 관리 (리렌더링 간 유지)
2. body 변경 시 `computeSectionBoundaries(body)`로 현재 section 경계 계산
3. `cache.updateSectionOrder(newIds)` — 순서 변경 시 전체 캐시 무효화 + 전체 재렌더링 요청
4. 순서 변경 없을 때: 이전 body 대비 변경된 section만 Worker에 재요청
5. **Pitfall 4 대응:** section별 requestId를 `Map<sectionId, requestId>`로 관리
   - Worker 응답 시 `requestIdMap.get(sectionId) === response.id` 조건 검증 후 캐시 갱신
6. Worker 에러 시 메인 스레드 fallback (`renderMarkdownPreview`)
7. 300ms debounce 적용 (기존 `useMarkdownPreview`와 동일)

`prevBodyRef = useRef<string>('')` — body가 변경될 때 이전 section 경계와 비교하여 변경된 section 식별
