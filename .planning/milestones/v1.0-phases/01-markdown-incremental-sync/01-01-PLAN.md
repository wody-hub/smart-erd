// client/src/lib/markdown-section-index.ts — 이 플랜에서 생성

export interface SectionBoundary {
  /** heading이 시작하는 body 내 character offset (root section은 0) */
  startOffset: number;
  /** 다음 section의 startOffset (마지막 section은 body.length) */
  endOffset: number;
  /** 안정적 section ID: slug 기반, 같은 slug 충돌 시 '-1'/'-2' 접미사 */
  id: string;
  /** heading level (1~6). root section은 0 */
  level: number;
  /** heading 텍스트 (root section은 '') */
  text: string;
}

/**
 * markdown body에서 section 경계 배열을 계산한다.
 * heading 이전 영역은 id='root', level=0으로 처리한다.
 *
 * @param body markdown 본문 (frontmatter 제외, \r\n은 \n으로 정규화된 상태)
 * @returns section 경계 배열 (순서 보장)
 */
export function computeSectionBoundaries(body: string): SectionBoundary[];

/**
 * 변경된 offset 범위가 속하는 section ID를 반환한다.
 *
 * @param boundaries computeSectionBoundaries 결과
 * @param changeStart 변경 시작 offset (inclusive)
 * @param changeEnd 변경 종료 offset (exclusive)
 * @returns 단일 section 내 변경이면 sectionId, 경계를 넘으면 null
 */
export function findAffectedSection(
  boundaries: SectionBoundary[],
  changeStart: number,
  changeEnd: number,
): string | null;
```

/**
 * SectionBoundary — heading 단위 section 경계 정보.
 */
export interface SectionBoundary {
  startOffset: number;
  endOffset: number;
  id: string;
  level: number;
  text: string;
}

export function computeSectionBoundaries(body: string): SectionBoundary[] {
  // 1. 정규식으로 heading 위치 추출: /^(#{1,6})\s+(.+)$/gm
  // 2. heading 이전 content가 있으면 root 섹션 prepend
  // 3. heading 텍스트 → slug: 소문자 변환, 알파숫자/공백만 유지, 공백→'-', 다중 '-'→단일
  // 4. slug 충돌 카운터: Map<string, number> — 첫 등장은 그대로, N번째 중복은 '-{N-1}' 접미사
  // 5. 각 section의 endOffset = 다음 section의 startOffset (마지막은 body.length)
  // 6. heading 이전 content가 없고 첫 요소가 heading인 경우 root 섹션 생략
  //    (root 섹션은 content가 있을 때만 추가)
}

export function findAffectedSection(
  boundaries: SectionBoundary[],
  changeStart: number,
  changeEnd: number,
): string | null {
  // 변경 범위 [changeStart, changeEnd)가 단일 section 내에 완전히 포함되면 해당 id 반환
  // 경계를 넘으면 null
  // 빈 변경 (changeStart === changeEnd)도 포함되는 section id 반환
}
```

**중요 구현 주의사항 (RESEARCH.md Pitfall 1 반영):**
- ID 생성 시 index 절대 사용 금지 — `heading-{index}-{slug}` 패턴 금지
- slug 전용 ID: `slugify(text)` + 충돌 시 `-1`, `-2` 접미사
- slugify: `text.toLowerCase().replace(/[^\w\s-]/g, '').replace(/\s+/g, '-').replace(/-+/g, '-').replace(/^-|-$/g, '')`
- 빈 slug (특수문자만인 heading) → `'section'` 기본값 사용 후 충돌 접미사 적용
