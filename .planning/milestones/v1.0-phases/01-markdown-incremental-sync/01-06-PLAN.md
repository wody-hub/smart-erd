```typescript
/** fenced code block(``` 또는 ~~~) 영역의 [start, end) 범위 배열 */
function findFencedCodeRanges(body: string): Array<{ start: number; end: number }> {
  const FENCE_PATTERN = /^(`{3,}|~{3,}).*$/gm;
  const ranges: Array<{ start: number; end: number }> = [];
  let openFence: { marker: string; start: number } | null = null;

  let match: RegExpExecArray | null;
  while ((match = FENCE_PATTERN.exec(body)) !== null) {
    const marker = match[1]!;
    if (!openFence) {
      openFence = { marker: marker[0]!.repeat(marker.length), start: match.index };
    } else if (
      marker[0] === openFence.marker[0] &&
      marker.length >= openFence.marker.length &&
      match[0]!.trim() === marker // closing fence는 info string 없음
    ) {
      ranges.push({ start: openFence.start, end: match.index + match[0]!.length });
      openFence = null;
    }
  }
  // unclosed fence: 나머지 전체를 code block으로
  if (openFence) {
    ranges.push({ start: openFence.start, end: body.length });
  }
  return ranges;
}
```

**문제:** `SectionPreviewCache.updateSectionOrder()`가 순서 변경 여부만 추적하고, 이전 section 순서에 있었지만 새 순서에 없는 section의 HTML 캐시를 정리하지 않는다. 장시간 편집 + heading 삭제/이름 변경 시 stale 캐시가 누적된다.

**수정 — `updateSectionOrder` 메서드에 GC 로직 추가:**

기존 `updateSectionOrder(newOrder)` 메서드에서, 순서 변경이 감지된 경우 `htmlCache`에서 `newOrder`에 없는 키를 삭제한다:

```typescript
updateSectionOrder(newOrder: string[]): boolean {
  const changed =
    newOrder.length !== this.sectionOrder.length ||
    newOrder.some((id, i) => id !== this.sectionOrder[i]);
  if (changed) {
    this.sectionOrder = [...newOrder];
    // GC: 현재 문서에 없는 section ID의 캐시 정리
    const activeIds = new Set(newOrder);
    for (const cachedId of this.htmlCache.keys()) {
      if (!activeIds.has(cachedId)) {
        this.htmlCache.delete(cachedId);
      }
    }
  }
  return changed;
}
```
